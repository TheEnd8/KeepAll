// Google Apps Script: AI 智慧管家 <-> Google Keep 雙向同步橋接器
// 將資料同步儲存至 Google Drive 專屬 Keep 同步資料夾與 Google Tasks/Keep 橋接表
const SYNC_FILE_NAME = 'AI_Knowledge_Keep_Bridge.json';

function doGet(e) {
  const file = getOrCreateBridgeFile();
  const content = file.getBlob().getDataAsString() || '[]';
  return ContentService.createTextOutput(content).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const payload = JSON.parse(e.postData.contents);
  const file = getOrCreateBridgeFile();
  let items = JSON.parse(file.getBlob().getDataAsString() || '[]');
  
  if (payload.action === 'upsert' && payload.item) {
    const idx = items.findIndex(i => i.id === payload.item.id);
    if (idx >= 0) items[idx] = payload.item;
    else items.unshift(payload.item);
  } else if (payload.action === 'batch_upsert' && Array.isArray(payload.items)) {
    payload.items.forEach(newItem => {
      const idx = items.findIndex(i => i.id === newItem.id);
      if (idx >= 0) items[idx] = newItem;
      else items.unshift(newItem);
    });
  }
  file.setContent(JSON.stringify(items, null, 2));
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok', count: items.length }))
    .setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateBridgeFile() {
  const files = DriveApp.getFilesByName(SYNC_FILE_NAME);
  if (files.hasNext()) return files.next();
  return DriveApp.createFile(SYNC_FILE_NAME, '[]', MimeType.PLAIN_TEXT);
}
