Google Keep 4 通道雙向整合與診斷中心

突破個人帳號無官方 Keep REST API 的限制，提供手機直傳、批次匯出、雲端橋接與反向還原

技術架構說明（個人 @gmail.com 帳號）：

Google 官方僅對企業付費網域管理員開放 Keep REST API，一般個人帳號無法透過背景 API 靜默寫入 Keep 伺服器。因此本系統為您提供以下 4 種在 Android 手機與電腦皆可 100% 順利運作的同步通道：

方式 1 · 手機最推薦 Android 原生「分享至 Keep」 & 網頁版快速貼上 0 筆待同步

點擊卡片底部的 Keep 按鈕，Android 手機會直接彈出系統分享面板，點選 「Keep 記事」 圖示即可原生建立含標題、所有網址與 #類別標籤 的筆記；電腦端則自動複製並開啟 Keep 網頁版。

方式 2 · 離線打包 一鍵匯出 Google Keep 專用批次檔（純文字 / Markdown / JSON）

將所有資料（或已勾選資料）整理為保留完整 #標籤 與時間戳記的標準格式，方便整批備份或貼入 Keep。

透過 Google Apps Script 自動將筆記寫入 Google 試算表備份，並在 Google Drive 建立「Keep_待匯入筆記」文件（在電腦開啟 keep.google.com 或 Gmail 右側 Keep 側邊欄，點選「將文件複製到 Keep」即可完成雲端匯入）。

部署步驟：開啟 script.google.com → 貼上以下程式碼 → 部署為「網頁應用程式（執行身分：我；誰可以存取：所有人）」 
// Google Apps Script: AI 智慧管家 <-> Google Sheets & Google Docs (Keep_待匯入筆記)
const SHEET_NAME = 'AI智慧管家_Keep同步庫';
const DOC_NAME = 'Keep_待匯入筆記';

function doGet(e) {
  try {
    const action = (e && e.parameter && e.parameter.action) || 'pull';
    if (action === 'ping') {
      return jsonResponse({ status: 'ok', httpCode: 200, message: 'GAS 連線診斷通過！已準備好寫入 Google 試算表與「Keep_待匯入筆記」文件。', time: new Date().toISOString() });
    }
    const sheet = getOrCreateSheet();
    const rows = sheet.getDataRange().getValues();
    const items = [];
    for (let i = 1; i < rows.length; i++) {
      if (!rows[i][0]) continue;
      items.push({
        id: String(rows[i][0]),
        title: String(rows[i][1] || ''),
        text: String(rows[i][2] || ''),
        labels: String(rows[i][3] || '').split(',').map(s => s.trim()).filter(Boolean),
        url: String(rows[i][4] || ''),
        reminder: rows[i][5] ? JSON.parse(rows[i][5]) : { time: '', location: '', context: '' },
        updatedAt: Number(rows[i][6]) || Date.now()
      });
    }
    return jsonResponse({ status: 'ok', items: items });
  } catch (err) {
    return jsonResponse({ status: 'error', message: err.toString() });
  }
}

function doPost(e) {
  try {
    const raw = e.postData && e.postData.contents ? e.postData.contents : '{}';
    const payload = JSON.parse(raw);
    const sheet = getOrCreateSheet();
    const incoming = payload.action === 'batch_upsert' && Array.isArray(payload.items)
      ? payload.items
      : (payload.item ? [payload.item] : []);

    const existingRows = sheet.getDataRange().getValues();
    const idRowMap = {};
    for (let i = 1; i < existingRows.length; i++) {
      if (existingRows[i][0]) idRowMap[String(existingRows[i][0])] = i + 1;
    }

    incoming.forEach(item => {
      const rowData = [
        item.id,
        item.title || '',
        item.text || '',
        Array.isArray(item.labels) ? item.labels.join(', ') : '',
        item.url || '',
        JSON.stringify(item.reminder || {}),
        item.updatedAt || Date.now()
      ];
      if (idRowMap[item.id]) {
        sheet.getRange(idRowMap[item.id], 1, 1, 7).setValues([rowData]);
      } else {
        sheet.appendRow(rowData);
      }
    });

    updateKeepBridgeDoc(sheet);
    return jsonResponse({ status: 'ok', syncedCount: incoming.length });
  } catch (err) {
    return jsonResponse({ status: 'error', message: err.toString() });
  }
}

function updateKeepBridgeDoc(sheet) {
  const files = DriveApp.getFilesByName(DOC_NAME);
  const doc = files.hasNext() ? DocumentApp.openById(files.next().getId()) : DocumentApp.create(DOC_NAME);
  const body = doc.getBody();
  body.clear();
  body.appendParagraph('📌 Keep_待匯入筆記（由 AI 智慧多媒體知識與靈感管家自動同步）').setHeading(DocumentApp.ParagraphHeading.HEADING1);
  body.appendParagraph('使用方式：在電腦開啟 keep.google.com 或 Gmail 右側 Keep 側邊欄，開啟本文件後點選「將文件複製到 Keep」即可。\n');
  const rows = sheet.getDataRange().getValues();
  for (let i = rows.length - 1; i >= 1; i--) {
    body.appendParagraph('【' + rows[i][1] + '】').setHeading(DocumentApp.ParagraphHeading.HEADING2);
    body.appendParagraph(rows[i][2] + '\n────────────────────\n');
  }
  doc.saveAndClose();
}

function getOrCreateSheet() {
  const files = DriveApp.getFilesByName(SHEET_NAME);
  const ss = files.hasNext() ? SpreadsheetApp.openById(files.next().getId()) : SpreadsheetApp.create(SHEET_NAME);
  const sheet = ss.getSheets()[0];
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['ID', '標題', 'Keep格式化內文', '類別標籤', '網址', '提醒JSON', '更新時間戳']);
  }
  return sheet;
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
